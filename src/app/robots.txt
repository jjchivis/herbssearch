# HerbsSearch.com
# Block Meta AI crawlers (also blocked by the Vercel Firewall in vercel.json).
# meta-externalads is deliberately NOT blocked, so Herbs Search can advertise
# through Meta later; facebookexternalhit (Facebook/Instagram link previews)
# stays allowed.

User-agent: meta-externalagent
Disallow: /

User-agent: meta-externalfetcher
Disallow: /

User-agent: meta-webindexer
Disallow: /

# Link previews when someone shares a Herbs Search page on Facebook or Instagram
User-agent: facebookexternalhit
Allow: /

# General crawler permissions (Googlebot, Bingbot and others)
User-agent: *
Allow: /
Disallow: /api/
Disallow: /admin/
# Library search results are endless duplicate pages; herb, topic and filter
# pages stay crawlable.
Disallow: /herbs?q=

# XML Sitemap
Sitemap: https://www.herbssearch.com/sitemap.xml
